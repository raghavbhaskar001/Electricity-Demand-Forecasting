

use $data/Table_A1_Final.dta, replace

gen dummy=.		// WILL BE USED IN TESTING SUBSAMPLE
gen alwaysoften=.	// WILL BE USED TO SUBSET PEOPLE WHO TAKE ACTION ALWAYS OR OFTEN

foreach i of varlist unplug_appliances buy_energy_efficient turnoff_ac switchoff_lights change_appliance_setting{

local label : variable label `i'

// HOW OFTEN THEY TAKE ACTION
quietly replace dummy=(`i'=="Always")
quietly ttest dummy, by(subsample)
local meanalwayssub=round(`r(mu_2)',.0001)*100
local meanalwayselse=round(`r(mu_1)',.0001)*100
local palways=round(`r(p)',.001)

quietly replace dummy=(`i'=="Often")
quietly ttest dummy, by(subsample)
local meanoftensub=round(`r(mu_2)',.0001)*100
local meanoftenelse=round(`r(mu_1)',.0001)*100
local poften=round(`r(p)',.001)


// MOTIVATIONS
// MOTIVATION 7 IS MONEY, MOTIVATION 8 IS HABIT, MOTIVATION 10 IS HEALTH


quietly replace alwaysoften=(`i'=="Always" | `i'=="Often")

quietly replace dummy=(regexm(mot_`i',"7"))
quietly replace dummy=. if alwaysoften==0
quietly ttest dummy, by(subsample)

local mean7sub=round(`r(mu_2)',.0001)*100
local mean7else=round(`r(mu_1)',.0001)*100
local p7=round(`r(p)',.001)

quietly replace dummy=(regexm(mot_`i',"8"))
quietly replace dummy=. if alwaysoften==0
quietly ttest dummy, by(subsample)

local mean8sub=round(`r(mu_2)',.0001)*100
local mean8else=round(`r(mu_1)',.0001)*100
local p8=round(`r(p)',.001)

quietly replace dummy=(regexm(mot_`i',"10"))
quietly replace dummy=. if alwaysoften==0
quietly ttest dummy, by(subsample)

local mean10sub=round(`r(mu_2)',.0001)*100
local mean10else=round(`r(mu_1)',.0001)*100
local p10=round(`r(p)',.001)

display "`label'&`meanalwayssub'&`meanalwayselse'&`meanoftensub'&`meanoftenelse'&`mean7sub'&`mean7else'&`mean8sub'&`mean8else'&`mean10sub'&`mean10else'"
display "&(`palways')&&(`poften')&&(`p7')&&(`p8')&&(`p10')" 

}
